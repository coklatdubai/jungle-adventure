Connect to Google Sheets

Follow these steps to automatically record scores:

  1. Open your Google Sheet.
  2. Go to Extensions > Apps Script.
  3. Delete any existing code and paste the code below.
  4. Click **Deploy > New deployment**.
  5. Click the gear icon, choose **Web app**.
  6. For "Who has access", select **Anyone**.
  7. Click **Deploy**, authorize the script, and copy the **Web app URL**.
  8. IMPORTANT: Paste that URL into the `googleSheetWebhookUrl` variable in this file's script.
function doPost(e){try{var t=SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Sheet1"),a=JSON.parse(e.postData.contents),s=a.name,o=a.studentClass,n=a.correctScore,c=a.incorrectScore,i=new Date;var r=t.getRange(1,1,1,5).getValues()[0];"Timestamp"!==r[0]||"Name"!==r[1]||"Class"!==r[2]||"Correct Score"!==r[3]||"Incorrect Score"!==r[4]||t.getRange("A1:E1").setValues([["Timestamp","Name","Class","Correct Score","Incorrect Score"]]).setFontWeight("bold"),t.appendRow([i,s,o,n,c]);return ContentService.createTextOutput(JSON.stringify({status:"success"})).setMimeType(ContentService.MimeType.JSON)}catch(t){return ContentService.createTextOutput(JSON.stringify({status:"error"})).setMimeType(ContentService.MimeType.JSON)}}
