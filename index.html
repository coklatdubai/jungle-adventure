### Part 1: Universal Google Apps Script (`Code.gs`) 1. Open your Google Spreadsheet and click **Extensions > Apps Script** (always open it from inside the Sheet so it is linked to that file). 2. Replace everything in `Code.gs` with the script below and click **Save**: ```javascript function doGet(e) { return handleRequest(e); } function doPost(e) { return handleRequest(e); } function handleRequest(e) { const lock = LockService.getScriptLock(); lock.tryLock(10000); try { const ss = SpreadsheetApp.getActiveSpreadsheet(); if (!ss) { return ContentService.createTextOutput( "ERROR: This script is not attached to a Google Sheet. Open your Google Sheet and click Extensions > Apps Script." ); } const sheet = ss.getSheets()[0]; // Create headers automatically if Sheet1 is empty if (sheet.getLastRow() === 0) { sheet.appendRow([ "Timestamp", "Explorer Name", "Class", "Activity / Station", "Score", "Correct Matches", "Stations Cleared", "Gems", "Total Mistakes", "Accuracy", "Station 1 Errors", "Station 2 Errors", "Station 3 Errors", "Mistake Details (Expected != Picked)", "Full Status" ]); sheet.setFrozenRows(1); } // Read both URL query parameters AND JSON body so nothing ever mismatches let p = (e && e.parameter) ? Object.assign({}, e.parameter) : {}; if (e && e.postData && e.postData.contents) { try { const json = JSON.parse(e.postData.contents); p = Object.assign(p, json); } catch (ignore) {} } // If opened in a browser tab with zero data, just confirm connection is live if (!p.name && !p.activity) { return ContentService.createTextOutput( "CONNECTED! Your Web App is live and linked to sheet: " + sheet.getName() ); } sheet.appendRow([ new Date(), p.name || "Explorer", p.studentClass || p.class || "", p.activity || "", Number(p.score || 0), Number(p.correctAnswers || 0), Number(p.questionsCompleted || 0), Number(p.gems || 0), Number(p.wrongAnswers || 0), p.accuracy || "100%", Number(p.s1Mistakes || 0), Number(p.s2Mistakes || 0), Number(p.s3Mistakes || 0), p.mistakeDetails || "None", p.status || "" ]); return ContentService .createTextOutput("SUCCESS! Row recorded in sheet: " + sheet.getName()) .setMimeType(ContentService.MimeType.TEXT); } catch (err) { return ContentService .createTextOutput("SCRIPT ERROR: " + err.toString()) .setMimeType(ContentService.MimeType.TEXT); } finally { lock.releaseLock(); } } ``` 3. Click **Deploy > Manage deployments > Edit (pencil icon)**. 4. Under **Version**, choose **New version** (critical step!). 5. Under **Execute as**, choose **Me**, and under **Who has access**, choose **Anyone** (*Note: if your school or government Google account only shows "Anyone within organization" and hides "Anyone", create the Sheet in a personal `@gmail.com` account so student browsers don't get blocked by organization login walls*). 6.
